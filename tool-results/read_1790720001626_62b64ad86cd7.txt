    71→        const conf = typeof out.confidence === "number" ? pct(out.confidence) : "—";
    72→        const txt = typeof out.transcript === "string" ? out.transcript.slice(0, 80) : JSON.stringify(out);
    73→        summary = `Transcription ${conf} confidence · ${txt}…`;
    74→      } else if (agentId === "qa") {
    75→        const qs = typeof out.quality_score === "number" ? out.quality_score : "—";
    76→        const rs = typeof out.risk_severity === "string" ? out.risk_severity : "—";
    77→        const fc = Array.isArray(out.findings) ? out.findings.length : 0;
    78→        summary = `QA score ${qs} · risk ${rs} · ${fc} findings`;

    80→        const an = typeof out.assigned_user_name === "string" ? out.assigned_user_name : "—";
    81→        summary = `Allocated to ${an}`;
    82→      } else if (agentId === "delivery") {
    83→        summary = typeof out.method === "string" ? `Delivered via ${out.method}` : "Delivered";
    84→      } else {
    85→        summary = `${label} succeeded · ${run.duration_ms}ms · ${run.cost_usd.toFixed(4)} cost`;
    86→      }

   372→  async function handleAction(action: string, label: string) {
   373→    try {
   374→      const res = await fetch("/api/medflow/job-status", {
   375→        method: "POST",
   376→        headers: { "Content-Type": "application/json" },
   377→        body: JSON.stringify({ jobId: job!.id, action, actorName: "Dana Whitfield" }),
   378→      });
   379→      if (!res.ok) throw new Error(`${res.status}`);
   380→      toast.success(`${job!.jobNumber} → ${label}`);
   381→      onAdvanced();
   382→      onClose();
   383→    } catch (e) {
   384→      toast.error(`Failed: ${(e as Error).message}`);
   385→    }
   386→  }
   387→}

   389→function MetaTile({ label, value }: { label: string; value: string }) {
   390→  return (
   391→    <div className="rounded-lg border border-border bg-card p-2.5">
   392→      <p className="text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">{label}</p>

   398→function ConfidenceRing({ value }: { value: number }) {
   399→  const pctVal = Math.max(0, Math.min(1, value));
   400→  const r = 22;
   401→  const c = 2 * Math.PI * r;
   402→  const off = c * (1 - pctVal);
   403→  return (
   404→    <svg width="64" height="64" viewBox="0 0 64 64">
   405→      <circle cx="32" cy="32" r={r} fill="none" stroke="#e2eee9" strokeWidth="6" />
   406→      <circle
   407→        cx="32" cy="32" r={r} fill="none" stroke="#1ABC9C" strokeWidth="6"
   408→        strokeDasharray={c} strokeDashoffset={off}
   409→        strokeLinecap="round" transform="rotate(-90 32 32)"
   410→      />
   411→      <text x="32" y="36" textAnchor="middle" className="fill-foreground text-[11px] font-bold">
   412→        {Math.round(pctVal * 100)}%
   413→      </text>
   414→    </svg>
   415→  );
   416→}

   418→function StructuredDoc({ job }: { job: JobWithRelations }) {
   419→  let parsed: Record<string, string> | null = null;
   420→  try {
   421→    if (job.structuredDocJson) parsed = JSON.parse(job.structuredDocJson);
   422→  } catch { /* ignore */ }
   423→  const entities = parsed?.["entities"];
   424→  return (
   425→    <div className="space-y-3">
   426→      <Card>
   427→        <CardContent className="p-4">
   428→          {parsed ? (
   429→            <div className="prose prose-sm max-w-none">
   430→              {Object.entries(parsed).filter(([k]) => k !== "entities").map(([k, v]) => (
   431→                <div key={k} className="mb-3">
   432→                  <p className="text-[11px] font-bold uppercase tracking-wider text-secondary">{k}</p>
   433→                  <p className="mt-0.5 text-sm leading-relaxed text-foreground">{v}</p>
   434→                </div>
   435→              ))}
   436→            </div>
   437→          ) : (
   438→            <p className="text-sm text-muted-foreground">No structured document yet — run the Structuring Agent.</p>
   439→          )}
   440→        </CardContent>
   441→      </Card>
   442→      {entities && (
   443→        <Card>
   444→          <CardContent className="p-3">
   445→            <p className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground">Extracted entities</p>
   446→            <div className="mt-2 flex flex-wrap gap-1.5">
   447→              {(Array.isArray(entities) ? entities : []).map((e: unknown, i: number) => (
   448→                <Badge key={i} variant="outline" className="text-[10px]">
   449→                  {typeof e === "string" ? e : JSON.stringify(e)}
   450→                </Badge>
   451→              ))}
   452→              {!Array.isArray(entities) && typeof entities === "object" && entities && (
   453→                Object.entries(entities as Record<string, unknown>).map(([k, v]) => (
   454→                  <Badge key={k} variant="outline" className="text-[10px]">
   455→                    <span className="font-semibold text-secondary">{k}:</span> {String(v)}
   456→                  </Badge>
   457→                ))
   458→              )}
   459→            </div>
   460→          </CardContent>
   461→        </Card>
   462→      )}
   463→    </div>
   464→  );
   465→}

   467→function QaFindings({ job }: { job: JobWithRelations }) {
   468→  const review = job.qualityReviews?.[0];
   469→  let findings: Array<{ type?: string; severity?: string; detail?: string; suggestion?: string }> = [];
   470→  try {
   471→    findings = review ? JSON.parse(review.findingsJson) : [];
   472→  } catch { /* ignore */ }
   473→  const sevStyle = (sev: string) => RISK_SEVERITY_STYLES[sev as keyof typeof RISK_SEVERITY_STYLES] ?? RISK_SEVERITY_STYLES.informational;
   474→  return (
   475→    <div className="space-y-3">
   476→      <div className="grid grid-cols-2 gap-3 sm:grid-cols-4">
   477→        <MetaTile label="Quality score" value={review?.qualityScore?.toString() ?? "—"} />
   478→        <MetaTile label="Risk score" value={review?.riskScore?.toString() ?? "—"} />
   479→        <MetaTile label="Risk severity" value={review?.riskSeverity ?? "—"} />
   480→        <MetaTile label="Recommendation" value={review?.recommendedAction ?? "—"} />
   481→      </div>
   482→      {findings.length === 0 ? (
   483→        <div className="rounded-lg border border-dashed border-border py-8 text-center text-sm text-muted-foreground">
   484→          <ShieldCheck className="mx-auto mb-2 h-6 w-6 text-emerald-500" />
   485→          No QA findings — run the QA Agent to inspect.
   486→        </div>
   487→      ) : (
   488→        <ul className="space-y-2">
   489→          {findings.map((f, i) => {
   490→            const s = sevStyle(f.severity ?? "informational");
   491→            return (
   492→              <motion.li
   493→                key={i}
   494→                initial={{ opacity: 0, y: 6 }}
   495→                animate={{ opacity: 1, y: 0 }}
   496→                transition={{ duration: 0.25, delay: i * 0.04 }}
   497→                className="rounded-lg border border-border bg-card p-3"
   498→              >
   499→                <div className="flex items-center justify-between gap-2">
   500→                  <div className="flex items-center gap-2">
   501→                    <span className={`h-2 w-2 rounded-full ${s.dot}`} />
   502→                    <span className="text-[12px] font-semibold text-foreground">{f.type ?? "Finding"}</span>
   503→                  </div>
   504→                  <span className={`rounded-full border px-2 py-0.5 text-[10px] font-semibold ${s.badge}`}>{s.label}</span>
   505→                </div>
   506→                {f.detail && <p className="mt-1 text-[12px] text-muted-foreground">{f.detail}</p>}
   507→                {f.suggestion && (
   508→                  <p className="mt-1 text-[11px] italic text-foreground/70">
   509→                    <span className="font-semibold">Suggestion:</span> {f.suggestion}
   510→                  </p>
   511→                )}
   512→              </motion.li>
   513→            );
   514→          })}
   515→        </ul>
   516→      )}
   517→    </div>
   518→  );
