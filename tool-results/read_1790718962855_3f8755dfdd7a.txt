    14→
    15→Responsibilities:
    16→  1. Orchestrate 10 AI agents (intake-validation → reporting).
    17→  2. Abstract multi-database connectors (SQLite / PostgreSQL / Supabase /
    18→     MySQL / MongoDB) with simulated connection-test + switch endpoints.

    20→     payloads for downstream EHR delivery.

   565→def _agent_output(agent_id: str, job_id: str, correlation_id: str) -> dict[str, Any]:
   566→    """
   567→    Produce a realistic, agent-type-specific output payload.
   568→    The shapes are defined in the task spec — keep them faithful.
   569→    """

  1179→@app.post("/api/agents/{agent_id}/run")
  1180→async def run_agent(agent_id: str, body: AgentRunRequest) -> dict[str, Any]:
  1181→    """
  1182→    Simulate running an agent on a job. Sets the agent to `running`,
  1183→    sleeps 600-1500ms (simulated work), then produces a realistic,
  1184→    agent-type-specific output and records the run.
  1185→    """
  1186→    if agent_id not in AGENTS:
  1187→        raise HTTPException(
  1188→            status_code=404,
  1189→            detail={
  1190→                "error": "not_found",
  1191→                "message": f"Unknown agent_id '{agent_id}'",
  1192→                "correlation_id": _generate_correlation_id(),
  1193→            },
  1194→        )
  1195→
  1196→    agent = AGENTS[agent_id]
  1197→    correlation_id = body.correlation_id or _generate_correlation_id()
  1198→    started_at = _now_iso()
  1199→
  1200→    # Flip status to running.
  1201→    agent["status"] = "running"
  1202→
  1203→    # Simulate latency between 600 and 1500 ms.
  1204→    duration_ms = random.randint(600, 1500)
  1205→    await asyncio.sleep(duration_ms / 1000.0)
  1206→
  1207→    # ~5% chance of a simulated failure (with retry suggestion) — mostly succeed.
  1208→    # Skip failure chance for rule-engine agents on the seeded jobs to keep the
  1209→    # demo deterministic for the curl verification (transcription always succeeds).
  1210→    simulate_fail = (
  1211→        random.random() < 0.05
  1212→        and agent_id not in ("transcription",)
  1213→        and body.job_id in SEED_JOBS
  1214→    )
  1215→
  1216→    completed_at = _now_iso()
  1217→    cost_usd = _approx_cost_usd(agent_id, duration_ms)
  1218→
  1219→    if simulate_fail:
  1220→        # Failed run — increment failure count, append to recent_runs, do NOT
  1221→        # produce a typed output (return a failure envelope).
  1222→        agent["status"] = "failed"
  1223→        agent["failure_count"] += 1
  1224→        run_record = {
  1225→            "agent_id": agent_id,
  1226→            "job_id": body.job_id,
  1227→            "tenant_id": body.tenant_id,
  1228→            "correlation_id": correlation_id,
  1229→            "run_id": f"run-{uuid.uuid4().hex[:10]}",
  1230→            "status": "failed",
  1231→            "started_at": started_at,
  1232→            "completed_at": completed_at,
  1233→            "duration_ms": duration_ms,
  1234→            "cost_usd": cost_usd,
  1235→            "model": agent["model"],
  1236→            "prompt_version": agent["prompt_version"],
  1237→            "error": "Simulated transient failure",
  1238→            "retry_recommended": True,
  1239→        }
  1240→    else:
  1241→        # Success path — produce typed output, update agent counters.
  1242→        output = _agent_output(agent_id, body.job_id, correlation_id)
  1243→        agent["status"] = "succeeded"
  1244→        agent["success_count"] += 1
  1245→        agent["last_run_at"] = completed_at
  1246→        agent["last_duration_ms"] = duration_ms
  1247→        agent["last_cost_usd"] = cost_usd
  1248→        run_record = {
  1249→            "agent_id": agent_id,
  1250→            "job_id": body.job_id,
  1251→            "tenant_id": body.tenant_id,
  1252→            "correlation_id": correlation_id,
  1253→            "run_id": f"run-{uuid.uuid4().hex[:10]}",
  1254→            "status": "succeeded",
  1255→            "started_at": started_at,
  1256→            "completed_at": completed_at,
  1257→            "duration_ms": duration_ms,
  1258→            "cost_usd": cost_usd,
  1259→            "model": agent["model"],
  1260→            "prompt_version": agent["prompt_version"],
  1261→            "output": output,
  1262→        }
  1263→
  1264→    # Update agent's recent_runs (cap 5 for the per-agent list).
  1265→    agent["recent_runs"].append(
  1266→        {
  1267→            "job_id": body.job_id,
  1268→            "status": run_record["status"],
  1269→            "duration_ms": duration_ms,
  1270→            "cost_usd": cost_usd,
  1271→            "timestamp": completed_at,
  1272→            "correlation_id": correlation_id,
  1273→        }
  1274→    )
  1275→    agent["recent_runs"] = agent["recent_runs"][-5:]
  1276→
  1277→    # Update the global RUNS log (cap 100).
  1278→    RUNS.append(run_record)
  1279→    if len(RUNS) > 100:
  1280→        del RUNS[: len(RUNS) - 100]
  1281→
  1282→    # Reset agent status to idle once the run is done (so the next call starts fresh).
  1283→    agent["status"] = "failed" if simulate_fail else "idle"
  1284→    # But mark the most recent outcome via last_run fields — kept above.
  1285→
  1286→    logger.info(
  1287→        "agent run complete agent=%s job=%s status=%s duration_ms=%s",
  1288→        agent_id,
  1289→        body.job_id,
  1290→        run_record["status"],
  1291→        duration_ms,
  1292→    )
  1293→
  1294→    return run_record
